### 1.3 核心代码结构

```python
class HierarchicalMemoryManager:
    """分层记忆管理器 - 管理L1-L4四层记忆"""

    def __init__(self, conversation_id: str, user_id: str):
        # L1: 工作记忆（内存）
        self.l1_working: List[Dict] = []
        self.l1_max_size = 3  # 3轮对话

        # L2: 短期记忆（PostgreSQL）
        self.l2_short_term = ShortTermMemory(conversation_id)

        # L3: 长期记忆（Qdrant向量库）
        self.l3_long_term = LongTermMemory(conversation_id)

        # L4: 核心记忆（JSON文件）
        self.l4_core = CoreMemory(user_id)

    def build_memory_context(self, current_query: str) -> MemoryContext:
        """构建完整的记忆上下文（供LLM使用）"""
        return MemoryContext(
            working_memory=self.get_working_memory(),      # L1
            short_term=self.l2_short_term.get_context(),   # L2
            long_term=self.l3_long_term.retrieve(current_query),  # L3
            core_memory=self.l4_core.get_all()             # L4
        )
```

## 🎯 二、多模式推理系统

### 2.1 推理模式对比

| 模式 | 触发条件 | LLM调用次数 | 适用场景 | 响应时间 |
|------|----------|-------------|----------|----------|
| **quick** | 问候/短句 | 1次 | "你好""谢谢" | < 1s |
| **standard** | 一般问题 | 1次 | "Python怎么写循环" | 1-3s |
| **deep** | 复杂分析 | 3次+ | "分析这个方案的优缺点" | 5-10s |
| **plan** | 多步骤任务 | 2次+工具链 | "帮我规划北京3日游" | 10s+ |

### 2.2 模式选择逻辑

```python
class ReasoningModeManager:
    """推理模式管理器 - 自动选择最优推理模式"""

    def select_mode(self, query: str) -> ReasoningMode:
        """根据查询特征自动选择模式"""
        query_lower = query.lower()

        # quick模式：问候语且长度<20
        quick_keywords = ["你好", "嗨", "hello", "hi", "谢谢", "再见"]
        if any(kw in query_lower for kw in quick_keywords) and len(query) < 20:
            return ReasoningMode.QUICK

        # plan模式：包含计划/步骤关键词
        plan_keywords = ["计划", "步骤", "如何", "怎么做", "流程"]
        if any(kw in query_lower for kw in plan_keywords):
            return ReasoningMode.PLAN

        # deep模式：需要分析/比较/解释
        deep_keywords = ["为什么", "分析", "比较", "评价", "优缺点", "深入"]
        if any(kw in query_lower for kw in deep_keywords):
            return ReasoningMode.DEEP

        # 默认standard模式
        return ReasoningMode.STANDARD
```

### 3.2 工具定义示例

```python
class SearchTool(BaseTool):
    """搜索工具 - 使用Tavily API进行联网搜索"""

    name = "search"
    description = "搜索互联网获取最新信息"
    parameters = {
        "query": {
            "type": "string",
            "description": "搜索关键词"
        },
        "max_results": {
            "type": "integer",
            "default": 5
        }
    }

    async def execute(self, query: str, max_results: int = 5) -> ToolOutput:
        try:
            response = await self.tavily_client.search(
                query=query,
                max_results=max_results
            )
            return ToolOutput(
                success=True,
                result=self._format_results(response)
            )
        except Exception as e:
            return ToolOutput(
                success=False,
                error=str(e)
            )
```

### 4.2 输出脱敏

```python
class ContentFilter:
    """内容过滤器 - 输入检测 + 输出脱敏"""

    # PII识别模式
    PII_PATTERNS = {
        'phone': r'1[3-9]\d{9}',           # 手机号
        'id_card': r'\d{17}[\dXx]',        # 身份证号
        'bank_card': r'\d{16,19}',         # 银行卡号
        'email': r'[\w.-]+@[\w.-]+\.\w+',  # 邮箱
    }

    def check_output(self, content: str, sanitize: bool = True) -> SecurityResult:
        """检查并脱敏输出内容"""
        result = SecurityResult(is_safe=True)

        for pii_type, pattern in self.PII_PATTERNS.items():
            matches = re.finditer(pattern, content)
            for match in matches:
                result.risk_types.append(pii_type)
                if sanitize:
                    # 脱敏: 13812345678 → 138****5678
                    content = self._mask_pii(content, match, pii_type)

        result.sanitized_content = content
        return result
```
